        public void HTC()
        {
            var doc = Autodesk.AutoCAD.ApplicationServices.Application.DocumentManager.MdiActiveDocument;
            db = doc.Database;
            ed = doc.Editor;

            try
            {
                // 框选文字对象（TEXT 单行文字 + MTEXT 多行文字）
                var filter = new SelectionFilter(new TypedValue[] { new TypedValue(0, "TEXT,MTEXT") });
                var selRes = ed.GetSelection(filter);
                if (selRes.Status != PromptStatus.OK) return;

                // 圆直径与间距：固定值，不受 HDIA 等任何设置影响
                double diameter = 0.03;   // 圆直径 0.03mm（半径 0.015mm）
                double spacing = 0.06;    // 圆心间距 0.06mm

                double radius = diameter / 2.0;

                ed.WriteMessage("\n文字转网点: 圆直径 {0:F4}mm, 间距 {1:F3}mm (固定值，统一按微软雅黑字体转换轮廓)",
                    diameter, spacing);

                int textCount = 0;
                int totalCircles = 0;
                int failedTexts = 0;

                ObjectId[] textIds = selRes.Value.GetObjectIds();

                // ===== 第 1 遍事务：统一字体（非雅黑一律切换为独立雅黑样式）并提交，
                // 使实体显示范围/字体度量按最终字体彻底刷新后再进入第 2 遍。=====
                using (var trFont = db.TransactionManager.StartTransaction())
                {
                    foreach (ObjectId id in textIds)
                    {
                        var entFont = trFont.GetObject(id, OpenMode.ForRead, false) as Entity;
                        if (entFont == null) continue;
                        ObjectId textStyleId = ObjectId.Null;
                        if (entFont is DBText) textStyleId = ((DBText)entFont).TextStyleId;
                        else if (entFont is MText) textStyleId = ((MText)entFont).TextStyleId;
                        if (textStyleId.IsNull) continue;
                        var textStyle = trFont.GetObject(textStyleId, OpenMode.ForRead, false)
                            as TextStyleTableRecord;
                        if (textStyle == null) continue;
                        if (IsMicrosoftYaheiFont(textStyle.FileName, textStyle.Font.TypeFace))
                            continue;
                        bool bold = textStyle.Font.Bold;
                        bool upgraded = false;
                        try
                        {
                            entFont.UpgradeOpen();
                            upgraded = true;
                            ObjectId yaheiStyleId = EnsureMicrosoftYaheiStyle(trFont, bold);
                            if (!yaheiStyleId.IsNull)
                            {
                                if (entFont is DBText) ((DBText)entFont).TextStyleId = yaheiStyleId;
                                else if (entFont is MText) ((MText)entFont).TextStyleId = yaheiStyleId;
                            }
                        }
                        finally
                        {
                            if (upgraded) entFont.DowngradeOpen();
                        }
                    }
                    trFont.Commit();
                }

                // ===== 第 2 遍事务：全新打开实体读取最终字体下的几何数据，
                // 渲染轮廓并按文字实际显示范围生成网点。=====
                using (var tr = db.TransactionManager.StartTransaction())
                {
                    BlockTableRecord modelSpace = (BlockTableRecord)tr.GetObject(
                        SymbolUtilityServices.GetBlockModelSpaceId(db), OpenMode.ForWrite);

                    foreach (ObjectId id in textIds)
                    {
                        var ent = tr.GetObject(id, OpenMode.ForRead, false) as Entity;
                        if (ent == null) continue;

                        // 提取文字内容与旋转角
                        string textString = null;
                        double rotation = 0;
                        ObjectId textStyleId = ObjectId.Null;
                        bool isMText = false;
                        double textHeight = 0;
                        double lineSpacingFactor = 1.0;
                        double widthFactor = 1.0;
                        double oblique = 0;
                        AttachmentPoint attachment = AttachmentPoint.TopLeft;
                        TextHorizontalMode hMode = TextHorizontalMode.TextLeft;
                        TextVerticalMode vMode = TextVerticalMode.TextBase;
                        Point3d position = new Point3d(0, 0, 0);
                        Point3d alignmentPoint = new Point3d(0, 0, 0);

                        if (ent is DBText dbText)
                        {
                            textString = dbText.TextString ?? "";
                            rotation = dbText.Rotation;
                            textStyleId = dbText.TextStyleId;
                            textHeight = dbText.Height;
                            widthFactor = dbText.WidthFactor;
                            oblique = dbText.Oblique;
                            hMode = dbText.HorizontalMode;
                            vMode = dbText.VerticalMode;
                            position = dbText.Position;
                            alignmentPoint = dbText.AlignmentPoint;
                        }
                        else if (ent is MText mText)
                        {
                            textString = mText.Contents ?? "";
                            rotation = mText.Rotation;
                            textStyleId = mText.TextStyleId;
                            isMText = true;
                            textHeight = mText.TextHeight;
                            lineSpacingFactor = mText.LineSpacingFactor;
                            attachment = mText.Attachment;
                        }

                        if (string.IsNullOrEmpty(textString))
                        {
                            failedTexts++;
                            continue;
                        }

                        string fontFileName = null;
                        string bigFontFileName = null;
                        string typeFaceName = null;
                        bool fontBold = false;
                        bool fontItalic = false;
                        if (!textStyleId.IsNull)
                        {
                            var textStyle = tr.GetObject(textStyleId, OpenMode.ForRead, false)
                                as TextStyleTableRecord;
                            if (textStyle != null)
                            {
                                fontFileName = textStyle.FileName;
                                bigFontFileName = textStyle.BigFontFileName;
                                typeFaceName = textStyle.Font.TypeFace;
                                fontBold = textStyle.Font.Bold;
                                fontItalic = textStyle.Font.Italic;
                                if (isMText)
                                {
                                    widthFactor = textStyle.XScale;
                                    oblique = textStyle.ObliquingAngle;
                                }
                            }
                        }

                        // 文字实际显示范围（字体已在第 1 遍事务统一并提交，此处为最终字体下的真实范围）
                        Extents3d ext = ent.GeometricExtents;
                        if (ext.MaxPoint.X - ext.MinPoint.X < 1e-6 ||
                            ext.MaxPoint.Y - ext.MinPoint.Y < 1e-6)
                        {
                            failedTexts++;
                            continue;
                        }

                        // 统一按微软雅黑渲染字形轮廓
                        List<Polyline> outlines = BuildTextOutlines(
                            textString, ext, rotation, fontFileName, bigFontFileName,
                            typeFaceName, fontBold, fontItalic, isMText, textHeight,
                            lineSpacingFactor, widthFactor, oblique, attachment,
                            hMode, vMode, position, alignmentPoint);

                        if (outlines == null || outlines.Count == 0)
                        {
                            failedTexts++;
                            continue;
                        }

                        // 按奇偶填充规则在全部轮廓上生成蜂窝圆：
                        // 被奇数条轮廓包围的点=笔画（填充），偶数=孔洞（空心），
                        // 如"0"外圈与内圈之间的环带填充、中心空心
                        try
                        {
                            var gridPoints = GenerateHexGridText(outlines, spacing, true, radius);
                            foreach (var point in gridPoints)
                            {
                                Circle circle = new Circle(point, Vector3d.ZAxis, radius);
                                modelSpace.AppendEntity(circle);
                                tr.AddNewlyCreatedDBObject(circle, true);
                            }
                            totalCircles += gridPoints.Count;
                        }
                        finally
                        {
                            foreach (var poly in outlines)
                                poly.Dispose();
                        }
                        textCount++;
                    }
                    tr.Commit();
                }

                ed.WriteMessage("\n\n===== HTC命令完成 =====");
                ed.WriteMessage("\n处理文字: {0} 个", textCount);
                ed.WriteMessage("\n总计生成: {0} 个圆", totalCircles);
                ed.WriteMessage("\n圆直径: {0:F4}mm, 间距: {1:F3}mm", diameter, spacing);
                if (failedTexts > 0)
                    ed.WriteMessage("\n跳过文字(无法生成轮廓): {0} 个", failedTexts);
            }
            catch (Exception ex)
            {
                ed.WriteMessage("\n错误: " + ex.Message);
                if (ex.InnerException != null)
                    ed.WriteMessage("\n内部错误: " + ex.InnerException.Message);
            }
        }
